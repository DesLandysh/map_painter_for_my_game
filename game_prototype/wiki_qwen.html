smoothstep(a, b, x)

Плавная интерполяция с easing-функцией

const smoothstep = (a, b, x) => {
  const t = clamp((x - a) / (b - a), 0, 1);
  return t * t * (3 - 2 * t);
};

Алгоритм: Использует полином Эрмита для плавного перехода

expDamp(current, target, lambda, dt)

Экспоненциальное затухание для плавного приближения к цели

const expDamp = (current, target, lambda, dt) =>
  current + (target - current) * (1 - Math.exp(-lambda * dt));
ПараметрОписание
currentТекущее значение
targetЦелевое значение
lambdaСкорость затухания (больше = быстрее)
dtДельта времени (секунды)

dampAngle(current, target, lambda, dt)

Плавное вращение угла с учётом кратчайшего пути

const dampAngle = (current, target, lambda, dt) => {
  const delta = Math.atan2(
    Math.sin(target - current),
    Math.cos(target - current)
  );
  return current + delta * (1 - Math.exp(-lambda * dt));
};

Генерация шума

mulberry32(seed)

Детерминированный генератор псевдослучайных чисел

function mulberry32(seed) {
  let a = seed >>> 0;
  return function random() {
    a |= 0;
    a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

hash2(x, z)

Хеш-функция для 2D координат

function hash2(x, z) {
  const v = Math.sin(x * 127.1 + z * 311.7) * 43758.5453123;
  return v - Math.floor(v);
}

valueNoise(x, z)

Value Noise - процедурный шум для генерации ландшафта

function valueNoise(x, z) {
  const xi = Math.floor(x);
  const zi = Math.floor(z);
  const xf = x - xi;
  const zf = z - zi;
  const u = xf * xf * (3 - 2 * xf);
  const v = zf * zf * (3 - 2 * zf);
  const a = hash2(xi, zi);
  const b = hash2(xi + 1, zi);
  const c = hash2(xi, zi + 1);
  const d = hash2(xi + 1, zi + 1);
  return lerp(lerp(a, b, u), lerp(c, d, u), v);
}

Генерация карты

columnLocalPositions(tx, tz)

Вычисляет позиции 25 гексов в тайле 5×5

function columnLocalPositions(tx, tz) {
  const out = [];
  for (let row = 0; row < 5; row++) {
    const globalRow = tz * 5 + row;
    for (let col = 0; col < 5; col++) {
      const x = tileX(tx) + (col - 2) * HEX_SX +
                (globalRow % 2 ? HEX_SX * 0.5 : 0) - HEX_SX * 0.25;
      const z = tileZ(tz) + (row - 2) * HEX_SZ;
      out.push({ x, z });
    }
  }
  return out;
}

Параметры:

  • HEX_SX = SQRT3 * HEX_R - ширина гекса
  • HEX_SZ = 1.5 * HEX_R - высота гекса
  • TILE_W = 5 * HEX_SX - ширина тайла
  • TILE_D = 7.5 * HEX_R - глубина тайла

terrainForColumn(worldX, worldZ, tile)

Вычисляет высоту и цвет для каждого гекса на основе расстояния до центра острова

function terrainForColumn(worldX, worldZ, tile) {
  const island = tile.island;
  const dist = Math.hypot(worldX - island.x, worldZ - island.z);
  const q = clamp(dist / island.r, 0, 1);
  
  // Микро-рельеф (шум)
  const micro = valueNoise(worldX * 2.05 + island.seed * 4.1, worldZ * 2.05) * 0.58 +
                valueNoise(worldX * 6.3, worldZ * 6.3 - island.seed) * 0.22;
  
  // Поднятие пика
  const peakRange = island.r * 0.5;
  const peakT = clamp(1 - dist / peakRange, 0, 1);
  const peakRise = (3.15 + island.r * 0.24) * island.peak * Math.pow(peakT, 1.62);
  
  // Зоны по расстоянию
  if (q > 0.79) { /* песок */ }
  else if (q > 0.56) { /* трава */ }
  else if (q > 0.39) { /* холмы */ }
  else { /* скалы */ }
}

  // Управление курсом через WASD

canSail(x, z)

Проверяет, можно ли плыть в данной точке (нет ли суши)

function canSail(x, z) {
  const margin = 0.68;
  if (x < -MAP_W / 2 + margin || x > MAP_W / 2 - margin) return false;
  
  const r = 0.67;
  for (let i = 0; i < 10; i++) {
    const a = (i / 10) * TAU;
    if (isLandPoint(x + Math.cos(a) * r, z + Math.sin(a) * r)) return false;
  }
  return true;
}

Система ветра

getWindAt(x, z)

Возвращает вектор ветра в конкретной точке мира

function getWindAt(x, z) {
  const localOffset = (valueNoise(
    x * wind.noiseScale + 7.1,
    z * wind.noiseScale + 3.3
  ) - 0.5) * 2 * wind.noiseSpread;
  
  const angle = wind.angle + localOffset;
  const strength = wind.strength * localStrength;
  
  return {
    x: Math.sin(angle) * strength,
    z: Math.cos(angle) * strength,
    angle,
    strength
  };
}
ПараметрЗначениеОписание
noiseScale0.035Масштаб шумового поля
noiseSpread0.45Макс. отклонение (~25°)
driftFactor0.32Сила сноса лодки

updateWind(dt)

Плавная смена направления и силы ветра (сезонность)

function updateWind(dt) {
  wind.seasonTimer += dt;
  if (wind.seasonTimer > wind.seasonDuration) {
    wind.seasonTimer = 0;
    const turn = (Math.random() - 0.5) * Math.PI * 0.9;
    wind.targetAngle += turn;
    wind.targetStrength = 0.25 + Math.random() * 0.65;
    wind.seasonDuration = 40 + Math.random() * 40;
  }
  
  // Плавная интерполяция
  wind.angle += delta * (1 - Math.exp(-0.09 * dt));
  wind.strength += (target - current) * (1 - Math.exp(-0.12 * dt));
}

⚡ Анализ оптимизации

Вопросы и рекомендации по улучшению производительности

Генерация геометрии в реальном времени Высокий приоритет

Проблема: Кораблик, маяк, коттеджи и пальмы генерируются процедурно каждый раз при загрузке через функции createBoat(), createCottage(), createPalm().

  • Вопрос: Нужен ли перевод в 3D-модели (GLTF/OBJ)?
  • Ответ: ДА, если количество объектов > 50. Для 1-2 объектов процедурная генерация приемлема.
  • Рекомендация: Экспортировать в GLTF и загружать через GLTFLoader. Это снизит время инициализации на 30-50%.
  • Альтернатива: Кэшировать геометрию и переиспользовать через clone()

🔴 Гексовые столбики в квадратных тайлах Высокий приоритет

Проблема: Каждый тайл 5×5 содержит 25 гексов, но тайлы прямоугольные. Это создаёт "ступенчатые" границы островов.

  • Вопрос: Нужно ли заменять на чистую гексовую сетку?
  • Ответ: ДА, если важна визуальная точность. НО это увеличит сложность кода в 2-3 раза.
  • Текущее решение: Функция wobble в terrainForColumn создаёт иллюзию неровного берега
  • Рекомендация: Оставить как есть для прототипа. Для релиза - перейти на axial/cube координаты гексов
  • Формула перехода: x = q * (sqrt(3) * size + 3/2 * size), z = r * (3/2 * size)

🟡 Процедурная генерация vs предустановленные острова Средний приоритет

Проблема: Острова генерируются процедурно через ISLANDS массив с параметрами {x, z, r, seed, peak}

  • Вопрос: Нужна ли процедурная генерация?
  • Ответ: ЗАВИСИТ от целей:
    • Для фиксированной карты (Карибы) - НЕТ, лучше расставить вручную
    • Для roguelike/рандомных карт - ДА, оставить процедурную
  • Рекомендация: Гибридный подход - основные острова вручную, мелкие детали процедурно
  • Оптимизация: Кэшировать результат генерации в localStorage для повторных загрузок

🟡 InstancedMesh vs отдельные меши Средний приоритет

Текущее состояние: Океан и земля используют InstancedMesh (хорошо!)

  • Проблема: Деревья, камни, пальмы - отдельные Group объекты
  • Вопрос: Нужно ли переводить в InstancedMesh?
  • Ответ: ДА, если количество > 100. Сейчас ~50 деревьев - приемлемо
  • Рекомендация: Для > 200 объектов использовать InstancedMesh с setMatrixAt()
  • Пример: treeSpots.forEach((tree, i) => { treeDummy.position.set(...); trees.setMatrixAt(i, treeDummy.matrix); })

🟡 Шейдер океана - пересборка при смене ветра Средний приоритет

Проблема: В шейдере океана используются JS-переменные через template literal:

float windX = ${Math.sin(wind.angle).toFixed(3)};
float windZ = ${Math.cos(wind.angle).toFixed(3)};
  • Вопрос: Нужно ли использовать uniform-переменные?
  • Ответ: ДА, если ветер меняется часто. Сейчас шейдер пересобирается каждый кадр!
  • Рекомендация: Добавить uniform uWindDirection и обновлять через shader.uniforms.uWindDirection.value
  • Экономия: ~2-5 мс на кадр при частой смене ветра

Текущие параметры:

const NX = 100;  // тайлов по X
const NZ = 40;   // тайлов по Z
const HEX_R = 0.155;

  • Рекомендация: Адаптивный LOD - меньше тайлов вдали от лодки
  • Формула: if (distFromBoat > 50) skipTile;

🟢 Система частиц - фиксированный пул Низкий приоритет

Текущее состояние: particleCount = 180 с кольцевым буфером

  • Вопрос: Достаточно ли 180 частиц?
  • Ответ: ДА для текущего геймплея. НО можно динамически менять
  • Рекомендация: Адаптивное количество: particleCount = baseCount * qualitySetting
  • Оптимизация: Использовать BufferGeometry с setDrawRange() вместо скрытия частиц
